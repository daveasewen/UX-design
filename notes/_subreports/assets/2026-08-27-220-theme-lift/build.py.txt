# ---------------------------------------------------------------- COLOUR MATHS
# WCAG 2.x relative luminance + contrast ratio, and CIE L* (D65) lightness. Copied from the #220
# banked sources so the three pages cannot disagree about the same pair of colours.


def _srgb_lin(c):
    c = c / 255.0
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def hex2rgb(h):
    h = h.lstrip('#')
    return (int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16))


def lum(rgb):
    r, g, b = [_srgb_lin(v) for v in rgb]
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def ratio(a, b):
    la, lb = lum(a), lum(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)


def lstar(rgb):
    y = lum(rgb)
    f = y ** (1.0 / 3.0) if y > 216.0 / 24389.0 else (24389.0 / 27.0 * y + 16.0) / 116.0
    return 116.0 * f - 16.0


def rgbstr(h):
    r, g, b = hex2rgb(h)
    return 'rgb(%d,%d,%d)' % (r, g, b)

def measure(ground_hex, page_hex, ink_hex):
    return {
        'hex': ground_hex,
        'rgb': rgbstr(ground_hex),
        'ink': ratio(hex2rgb(ink_hex), hex2rgb(ground_hex)),
        'sep': ratio(hex2rgb(ground_hex), hex2rgb(page_hex)),
        'dl': lstar(hex2rgb(ground_hex)) - lstar(hex2rgb(page_hex)),
    }


BEFORE = {t: measure(BEFORE_HEX[t], DARK_PAGE[t], INK_REV[t]) for t in THEMES}
AFTER = {t: measure(AFTER_HEX[t], DARK_PAGE[t], INK_REV[t]) for t in THEMES}

# Dave is astigmatic and red/yellow are his unstable hues, so a warm drift inside a grey would be
# exactly the invisible defect. Three themes must be R=G=B. Supercharge's ramp IS warm by RULING
# (ADR-0014 binds `neutral/N -> color/warm/N`), so its grounds are asserted MONOTONE WARM instead —
# R >= G >= B, and refused outright if the red channel runs away from the blue by more than the
# ruled ramp's own widest step.

def spec(override=None, probe='', cap='darkgrey', slot='--bm-cap-dark-darkgrey'):
    """One capsule specimen with a caption ground selected.

    ⛔ `override` IS FOR **BEFORE** (and for the two drawn grey candidates) ONLY. It substitutes
    ONE value into the mint's own slot. AFTER passes nothing: it renders through the enacted
    rules, which is the only way this page can be evidence rather than a drawing of an intention."""
    st = ' style="%s:%s;"' % (slot, override) if override else ''
    return ('<div class="br-spec"%s data-probe="%s"><div class="bm-stage" data-type="gallery" '
            'data-page-bg="page" data-spacing="40" data-keylines="off" data-bento-bg="transparent" '
            'data-mode="bento" data-edge="square" data-rounding="capsule" data-cap-bg="%s">'

            '<div class="c-bento bm-wall bm-gallery cdl-wall" data-bento-role="gallery">'

def nums(m, theme, extra=()):
    inband = BAND['min'] <= m['dl'] <= BAND['max']
    rows = [
        ('Caption ground', '%s &middot; %s' % (m['rgb'], m['hex'])),
        ('Page ground', '%s &middot; %s' % (rgbstr(DARK_PAGE[theme]), DARK_PAGE[theme])),
        ('Ink on that ground', '%.2f:1 &mdash; %s the ruled %s:1 floor'
         % (m['ink'], 'CLEARS' if m['ink'] >= INK_FLOOR else 'BELOW', INK_FLOOR)),
        ('Lift off the page', '&Delta;L* %+.2f &nbsp;&middot;&nbsp; %.3f:1%s'
         % (m['dl'], m['sep'],
            ' &nbsp;&middot;&nbsp; IN THE BAND' if inband else
            (' &nbsp;&middot;&nbsp; BELOW THE PAGE' if m['dl'] < 0 else
             ' &nbsp;&middot;&nbsp; outside the band'))),
    ]
    rows += list(extra)
    out = ['<dl class="cdl-nums">']
    for k, v in rows:
        out.append('<div class="cdl-num"><dt class="t-cm-legal">%s</dt>'
                   '<dd class="t-cm-figure-6">%s</dd></div>' % (k, v))
    out.append('</dl>')
    return ''.join(out)


def lines(items):
    return ('<ul class="cdl-lines">'
            + ''.join('<li class="t-ed-body-small">%s</li>' % i for i in items) + '</ul>')
