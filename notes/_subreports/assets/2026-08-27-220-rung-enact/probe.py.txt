def _lin(c):
    c = c / 255.0
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4
def lum(rgb):
    r, g, b = [_lin(v) for v in rgb]
    return 0.2126*r + 0.7152*g + 0.0722*b
def ratio(a, b):
    la, lb = lum(a), lum(b); hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)
def lstar(rgb):
    y = lum(rgb)
    f = y ** (1/3.0) if y > 216/24389.0 else (24389/27.0*y + 16)/116.0
    return 116*f - 16
def prgb(s):
    n = [int(float(x)) for x in s.replace('rgba(','').replace('rgb(','').replace(')','').split(',')[:3]]
    return tuple(n)

  const m = f => { c.font = '40px ' + f; return +c.measureText('Handgloves 12345').width.toFixed(2); };

}"""

JS_SPECS = """() => {

      capBox: cap.getBoundingClientRect().toJSON(),

}"""

def patch_mode(im, x, y, w, h):
    a = np.array(im.crop((int(x), int(y), int(x+w), int(y+h))).convert('RGB'))
    a = a.reshape(-1, 3)
    vals, counts = np.unique(a, axis=0, return_counts=True)
    return tuple(int(v) for v in vals[counts.argmax()]), float(counts.max()) / len(a)
