# ---------------------------------------------------------------- COLOUR MATHS
# WCAG 2.x relative luminance + contrast ratio, and CIE L* (D65) lightness. Copied from the
# #220 ladder's banked source so the two pages cannot disagree about the same pair of colours.


def _srgb_lin(c):
    c = c / 255.0
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def hex2rgb(h):
    h = h.lstrip('#')
    return (int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16))


def lum(rgb):
    r, g, b = [_srgb_lin(v) for v in rgb]
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def ratio(a, b):
    la, lb = lum(a), lum(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)


def lstar(rgb):
    y = lum(rgb)
    f = y ** (1.0 / 3.0) if y > 216.0 / 24389.0 else (24389.0 / 27.0 * y + 16.0) / 116.0
    return 116.0 * f - 16.0


def rgbstr(h):
    r, g, b = hex2rgb(h)
    return 'rgb(%d,%d,%d)' % (r, g, b)

def measure(ground_hex, page_hex, ink_hex):
    return {
        'hex': ground_hex,
        'rgb': rgbstr(ground_hex),
        'ink': ratio(hex2rgb(ink_hex), hex2rgb(ground_hex)),
        'sep': ratio(hex2rgb(ground_hex), hex2rgb(page_hex)),
        'dl': abs(lstar(hex2rgb(ground_hex)) - lstar(hex2rgb(page_hex))),
    }


BEFORE = measure(BEFORE_HEX, PAGE_DARK, INK_DARK)

def lift(m):
    """-> the rung's separation as a multiple of the light capsule's. The light card is the only
    separation Dave has approved, so it is the unit — not a rule, a REFERENCE."""
    return (m['sep'] - 1.0) / (LIGHT_PARITY - 1.0)

def nums(m, extra=()):
    rows = [
        ('Caption ground', '%s &middot; %s' % (m['rgb'], m['hex'])),
        ('Page ground', '%s &middot; %s' % (rgbstr(PAGE_DARK), PAGE_DARK)),
        ('Ink on that ground', '%.2f:1 &mdash; %s the ruled %s:1 floor'
         % (m['ink'], 'CLEARS' if m['ink'] >= INK_FLOOR else 'BELOW', INK_FLOOR)),
        ('Ground against ground', '%.2f:1 &nbsp;&middot;&nbsp; &Delta;L* %.2f' % (m['sep'], m['dl'])),
        ('Against the light card', '%.2f&times; its separation' % lift(m)),
    ]
    rows += list(extra)
    out = ['<dl class="cdl-nums">']
    for k, v in rows:
        out.append('<div class="cdl-num"><dt class="t-cm-legal">%s</dt>'
                   '<dd class="t-cm-figure-6">%s</dd></div>' % (k, v))
    out.append('</dl>')
    return ''.join(out)


def lines(items):
    return ('<ul class="cdl-lines">'
            + ''.join('<li class="t-ed-body-small">%s</li>' % i for i in items) + '</ul>')
