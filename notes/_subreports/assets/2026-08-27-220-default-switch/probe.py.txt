def _lin(c):
    c = c / 255.0
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def lum(rgb):
    r, g, b = [_lin(v) for v in rgb]
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def ratio(a, b):
    la, lb = lum(a), lum(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)


def prgb(s):
    n = [int(float(x)) for x in
         s.replace('rgba(', '').replace('rgb(', '').replace(')', '').split(',')[:3]]
    return tuple(n)


def alpha(s):
    parts = s.replace('rgba(', '').replace('rgb(', '').replace(')', '').split(',')
    return float(parts[3]) if len(parts) > 3 else 1.0

def _h2t(h):
    h = h.lstrip('#')

def patch_mode(im, x, y, w, h):
    a = np.array(im.crop((int(x), int(y), int(x + w), int(y + h))).convert('RGB'))
    a = a.reshape(-1, 3)
    vals, counts = np.unique(a, axis=0, return_counts=True)
    return tuple(int(v) for v in vals[counts.argmax()]), float(counts.max()) / len(a)
