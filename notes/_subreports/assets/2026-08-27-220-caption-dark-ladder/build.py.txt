# ---------------------------------------------------------------- COLOUR MATHS
# WCAG 2.x relative luminance + contrast ratio, and CIE L* (D65) lightness.
# Every candidate here is a NEUTRAL grey (R==G==B), asserted below, so luminance is the
# linearised channel value. No warm ramp, no tint drift — Dave's problem hues are red and
# yellow and nothing on this page carries a hue at all.
def _srgb_lin(c):
    c = c / 255.0
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4

def hex2rgb(h):
    h = h.lstrip('#')
    return (int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16))

def lum(rgb):
    r, g, b = [_srgb_lin(v) for v in rgb]
    return 0.2126 * r + 0.7152 * g + 0.0722 * b

def ratio(a, b):
    la, lb = lum(a), lum(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)

def lstar(rgb):
    y = lum(rgb)
    f = y ** (1.0 / 3.0) if y > 216.0 / 24389.0 else (24389.0 / 27.0 * y + 16.0) / 116.0
    return 116.0 * f - 16.0

def rgbstr(h):
    r, g, b = hex2rgb(h)
    return 'rgb(%d,%d,%d)' % (r, g, b)

def solve_grey(target_ratio, page_hex):
    """The neutral grey whose contrast against `page_hex` is closest to target_ratio.
    ⛔ SEARCH UPWARD ONLY. A contrast ratio is symmetric — max over min — so a grey DARKER
    than the page satisfies the same ratio, and an unconstrained search returns it. The
    question here is always 'how much LIGHTER than the page', so the search starts one
    step above the page channel. (Caught at build time #220: the first run returned
    #0A0A0A for light parity, which is darker than the page it was meant to lift off.)
    DERIVED AT MINT TIME (s200-D1): the value is computed here, never carried as a literal."""
    page = hex2rgb(page_hex)
    best, bestd = None, 1e9
    for c in range(page[0] + 1, 256):
        d = abs(ratio((c, c, c), page) - target_ratio)
        if d < bestd:
            best, bestd = c, d
    assert best is not None, 'no grey above the page ground satisfies %s' % target_ratio
    return '#%02X%02X%02X' % (best, best, best)

def spec(cap_bg, override=None, probe=''):
    """One capsule specimen. `override` substitutes ONE value into the caption-ground slot;
    nothing else about the card changes between one rung and the next."""
    st = ''
    if override:
        var = '--bm-darkgrey' if cap_bg == 'darkgrey' else '--bm-grey'
        st = ' style="%s:%s;"' % (var, override)
    return ('<div class="br-spec"%s data-probe="%s"><div class="bm-stage" data-type="gallery" '
            'data-page-bg="page" data-spacing="40" data-keylines="off" data-bento-bg="transparent" '
            'data-mode="bento" data-edge="square" data-rounding="capsule" data-cap-bg="%s">'
            '<div class="bm-page-ground"><div class="bm-pane" data-pane="gallery">'
            '<div class="c-bento bm-wall bm-gallery cdl-wall" data-bento-role="gallery">'
            '<div class="c-bento__grid">%s</div></div></div></div></div></div>'
            % (st, probe, cap_bg, TILE))

def dark_cell(r, probe):
    return ('<div class="cdl-cell"><div class="cdl-panel" data-apollo-theme="console">'
            '<div class="cdl-panel-bd" data-theme="dark">%s</div></div></div>'
            % spec(r['capbg'], r.get('override'), probe))

def nums(r):
    ink_ok = r['ink_c'] >= INK_FLOOR
    rows = [
      ('Caption ground', '%s &middot; %s' % (r['rgb'], r['hexv'])),
      ('Page ground', '%s &middot; %s' % (rgbstr(PAGE_DARK), PAGE_DARK)),
      ('Ink on that ground', '%.2f:1 &mdash; %s the ruled 4.5:1 floor'
          % (r['ink_c'], 'CLEARS' if ink_ok else 'BELOW')),
      ('Ground against ground', '%.2f:1 &nbsp;&middot;&nbsp; &Delta;L* %.2f' % (r['sep'], r['dl'])),
      ('Against light parity', '%s' % (
          'the same' if abs(r['sep'] - LIGHT_PARITY_RATIO) < 0.005
          else ('%.2f&times; the light capsule&rsquo;s separation' % (
                (r['sep'] - 1.0) / (LIGHT_PARITY_RATIO - 1.0)))))
    ]
    if r.get('token'):
        rows.append(('Token', r['token']))
    else:
        rows.append(('Token', 'NONE &mdash; this value does not exist in canon'))
    rows.append(('Ink token', r['ink_token']))
    out = ['<dl class="cdl-nums">']
    for k, v in rows:
        out.append('<div class="cdl-num"><dt class="t-cm-legal">%s</dt>'
                   '<dd class="t-cm-figure-6">%s</dd></div>' % (k, v))
    out.append('</dl>')
    return ''.join(out)

def rung(r, label, probe):
    cls = 'cdl-rung' + (' cdl-refused' if r.get('refused') else '')
    why = ''
    if r.get('refused'):
        why = ('<div class="cdl-why"><p class="t-ed-body-small"><b>Refused.</b> %s</p></div>'
               % r['refused'])
    prose = ('<p class="t-ed-body-small cdl-prose">%s</p>' % r['prose']) if r['prose'] else ''
    return ('<div class="%s">'
            '<p class="cdl-rung-name t-ed-body"><b>%s &mdash; %s</b></p>'
            '%s%s'
            '<div class="cdl-rung-bd">%s<div class="cdl-side">%s</div></div>'
            '</div>' % (cls, label, r['name'], prose, why, dark_cell(r, probe), nums(r)))

def lines(items):
    return ('<ul class="cdl-lines">'
            + ''.join('<li class="t-ed-body-small">%s</li>' % i for i in items) + '</ul>')
